  origin(origin, callback) {
    if (!origin) return callback(null, true);
    const isLocalOrigin = /^https?:\/\/(localhost|127\.0\.0\.1)(:\d+)?$/.test(origin);
    const isCodespacesOrigin = /^https:\/\/[a-zA-Z0-9-]+\.app\.github\.dev$/.test(origin);
    const isAllowed = config.corsOrigins.includes(origin) || (!config.isProduction && (isLocalOrigin || isCodespacesOrigin));

    return callback(new Error('CORS_ORIGIN_NOT_ALLOWED'));

}));

import Database from 'better-sqlite3';
import { mkdirSync } from 'node:fs';
import path from 'node:path';
import { SCHEMA_SQL } from './schema.js';

let database: Database.Database | null = null;

export function getDatabase(): Database.Database {
  if (database) return database;

  const dataDirectory = path.resolve(
    process.env.ZARA_DATA_DIR || path.join(process.cwd(), 'data')
  );

  mkdirSync(dataDirectory, { recursive: true });

  const databasePath = path.join(dataDirectory, 'zara.sqlite');

  database = new Database(databasePath);

  database.pragma('journal_mode = WAL');
  database.pragma('foreign_keys = ON');
  database.pragma('busy_timeout = 5000');

  database.exec(SCHEMA_SQL);

  return database;
}

export function closeDatabase(): void {
  if (database) {
    database.close();
    database = null;
  }
}

export interface ChatRequestRecord {
  request_id: string;
  thread_id: string;
  user_message_id: string;
  assistant_message_id: string | null;
  provider_id: string | null;
  model: string | null;
  status: 'running' | 'completed';
}

export function claimChatRequest(
  requestId: string,
  threadId: string,
  userMessageId: string,
): { claimed: boolean; record: ChatRequestRecord } {
  const db = getDatabase();
  // The additive table is created by the normal database schema migration.
  db.prepare(`DELETE FROM chat_requests WHERE status = 'running' AND created_at < datetime('now', '-10 minutes')`).run();

  const result = db.prepare(`
    INSERT OR IGNORE INTO chat_requests
      (request_id, thread_id, user_message_id, status)
    VALUES (?, ?, ?, 'running')
  `).run(requestId, threadId, userMessageId);

  const record = db.prepare('SELECT * FROM chat_requests WHERE request_id = ?')
    .get(requestId) as ChatRequestRecord;
  return { claimed: result.changes === 1, record };
}

export function completeChatRequest(
  requestId: string,
  assistantMessageId: string,
  providerId: string,
  model?: string,
): void {
  getDatabase().prepare(`
    UPDATE chat_requests
    SET assistant_message_id = ?, provider_id = ?, model = ?, status = 'completed'
    WHERE request_id = ? AND status = 'running'
  `).run(assistantMessageId, providerId, model ?? null, requestId);
}

export function releaseChatRequest(requestId: string): void {
  getDatabase().prepare("DELETE FROM chat_requests WHERE request_id = ? AND status = 'running'").run(requestId);
}

import { randomUUID } from 'node:crypto';
import { getDatabase } from '../database.js';

export type MessageRole = 'user' | 'assistant' | 'system';

export interface Message {
  id: string;
  thread_id: string;
  role: MessageRole;
  content: string;
  created_at: string;
}

export interface SearchMessageItem {
  id: string;
  thread_id: string;
  thread_title: string;
  role: MessageRole;
  content: string;
  created_at: string;
}

export interface SearchMessagesPaginatedParams {
  query: string;
  page?: number;
  limit?: number;
}

export interface SearchMessagesPaginatedResult {
  results: SearchMessageItem[];
  total: number;
  page: number;
  limit: number;
  totalPages: number;
}

export function createMessage(
  threadId: string,
  role: MessageRole,
  content: string,
  id = randomUUID(),
): Message {
  const db = getDatabase();
  const existing = getMessageById(id);
  if (existing) {
    if (existing.thread_id === threadId && existing.role === role && existing.content === content) {
      return existing;
    }
    throw new Error('MESSAGE_ID_CONFLICT');
  }

  const threadExists = db.prepare(
    'SELECT id FROM threads WHERE id = ?'
  ).get(threadId);

  if (!threadExists) {
    throw new Error('THREAD_NOT_FOUND');
  }

  if (!content.trim()) {
    throw new Error('MESSAGE_CONTENT_REQUIRED');
  }

  const save = db.transaction(() => {
    db.prepare(`
      INSERT INTO messages (id, thread_id, role, content)
      VALUES (?, ?, ?, ?)
    `).run(id, threadId, role, content);

    db.prepare(`
      UPDATE threads
      SET updated_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(threadId);
  });

  save();

  return getMessageById(id)!;
}

export function getMessageById(id: string): Message | undefined {
  return getDatabase()
    .prepare('SELECT * FROM messages WHERE id = ?')
    .get(id) as Message | undefined;
}

export function getMessagesByThread(threadId: string): Message[] {

    .all(threadId) as Message[];
}

export function deleteMessage(id: string): boolean {

export function searchMessages(
  query: string,
  limit = 50
): Message[] {
  const db = getDatabase();
  const term = query.trim();

  if (!term) return [];

  const safeLimit = Math.max(1, Math.min(limit, 100));

  try {
    const words = term
      .replace(/["*(){}[\]^:+-]/g, ' ')
      .trim()
      .split(/\s+/)
      .filter(Boolean);

    if (words.length === 0) return [];

    const matchQuery = words
      .map(word => `"${word.replace(/"/g, '""')}"`)
      .join(' AND ');

    return db.prepare(`
      SELECT m.*
      FROM messages_fts f
      JOIN messages m ON m.id = f.message_id
      WHERE messages_fts MATCH ?
      ORDER BY m.created_at DESC, m.rowid DESC
      LIMIT ?
    `).all(matchQuery, safeLimit) as Message[];
  } catch {
    return db.prepare(`
      SELECT * FROM messages
      WHERE content LIKE ?
      ORDER BY created_at DESC, rowid DESC
      LIMIT ?
    `).all(`%${term}%`, safeLimit) as Message[];
  }
}

function escapeLikePattern(value: string): string {
  return value.replace(/[\\%_]/g, '\\$&');
}

export function searchMessagesPaginated(
  params: SearchMessagesPaginatedParams
): SearchMessagesPaginatedResult {
  const rawPage =
    typeof params.page === 'number' && Number.isFinite(params.page)
      ? Math.trunc(params.page)
      : 1;

  const safePage = Math.max(1, rawPage);

  const rawLimit =
    typeof params.limit === 'number' && Number.isFinite(params.limit)
      ? Math.trunc(params.limit)
      : 20;

  const safeLimit = Math.max(1, Math.min(rawLimit, 100));

  const term =
    typeof params.query === 'string' ? params.query.trim() : '';

  if (!term) {
    return {
      results: [],
      total: 0,
      page: safePage,
      limit: safeLimit,
      totalPages: 0,
    };
  }

  const words = term
    .replace(/["*(){}[\]^:+-]/g, ' ')
    .trim()
    .split(/\s+/)
    .filter(Boolean);

  if (words.length === 0) {
    return {
      results: [],
      total: 0,
      page: safePage,
      limit: safeLimit,
      totalPages: 0,
    };
  }

  const db = getDatabase();
  const offset = (safePage - 1) * safeLimit;

  if (!Number.isSafeInteger(offset)) {
    throw new Error('INVALID_SEARCH_PAGE');
  }

  const matchQuery = words
    .map(word => `"${word.replace(/"/g, '""')}"`)
    .join(' AND ');

  try {
    const runFtsSearch = db.transaction(() => {
      const countRow = db.prepare(`
        SELECT COUNT(*) AS total
        FROM messages_fts f
        JOIN messages m ON m.id = f.message_id
        JOIN threads t ON t.id = m.thread_id
        WHERE messages_fts MATCH ?
      `).get(matchQuery) as { total: number } | undefined;

      const total = countRow?.total ?? 0;
      const totalPages = total === 0
        ? 0
        : Math.ceil(total / safeLimit);

      if (total === 0 || offset >= total) {
        return {
          results: [] as SearchMessageItem[],
          total,
          page: safePage,
          limit: safeLimit,
          totalPages,
        };
      }

      const results = db.prepare(`
        SELECT
          m.id,
          m.thread_id,
          t.title AS thread_title,
          m.role,
          m.content,
          m.created_at
        FROM messages_fts f
        JOIN messages m ON m.id = f.message_id
        JOIN threads t ON t.id = m.thread_id
        WHERE messages_fts MATCH ?
        ORDER BY m.created_at DESC, m.rowid DESC
        LIMIT ? OFFSET ?
      `).all(matchQuery, safeLimit, offset) as SearchMessageItem[];

      return {
        results,
        total,
        page: safePage,
        limit: safeLimit,
        totalPages,
      };
    });

    return runFtsSearch();
  } catch {
    const likePattern = `%${escapeLikePattern(term)}%`;

    const runLikeFallback = db.transaction(() => {
      const countRow = db.prepare(String.raw`
        SELECT COUNT(*) AS total
        FROM messages m
        JOIN threads t ON t.id = m.thread_id
        WHERE m.content LIKE ? ESCAPE '\'
      `).get(likePattern) as { total: number } | undefined;

      const total = countRow?.total ?? 0;
      const totalPages = total === 0
        ? 0
        : Math.ceil(total / safeLimit);

      if (total === 0 || offset >= total) {
        return {
          results: [] as SearchMessageItem[],
          total,
          page: safePage,
          limit: safeLimit,
          totalPages,
        };
      }

      const results = db.prepare(String.raw`
        SELECT
          m.id,
          m.thread_id,
          t.title AS thread_title,
          m.role,
          m.content,
          m.created_at
        FROM messages m
        JOIN threads t ON t.id = m.thread_id
        WHERE m.content LIKE ? ESCAPE '\'
        ORDER BY m.created_at DESC, m.rowid DESC
        LIMIT ? OFFSET ?
      `).all(likePattern, safeLimit, offset) as SearchMessageItem[];

      return {
        results,
        total,
        page: safePage,
        limit: safeLimit,
        totalPages,
      };
    });

    return runLikeFallback();
  }
}

import { getDatabase } from '../database.js';

export type ProviderStatus =
  | 'unconfigured'
  | 'pending_verification'
  | 'ready'
  | 'disabled';

export type SupportedProviderId = 'omniroute' | 'freellmapi';

export interface Provider {
  id: string;
  name: string;
  status: ProviderStatus;
  enabled: number;
  config: string;
  created_at: string;
  updated_at: string;
}

const DEFAULT_PROVIDERS = [
  { id: 'omniroute', name: 'OmniRoute', status: 'pending_verification' },
  { id: 'freellmapi', name: 'FreeLLMAPI', status: 'pending_verification' },
] as const;

export function isSupportedProviderId(id: string): id is SupportedProviderId {
  return id === 'omniroute' || id === 'freellmapi';
}

export function initializeProviders(): void {
  const db = getDatabase();
  const insert = db.prepare(`
    INSERT OR IGNORE INTO providers (id, name, status, enabled, config)
    VALUES (?, ?, ?, 0, '{}')
  `);
  const seed = db.transaction(() => {
    for (const provider of DEFAULT_PROVIDERS) {
      insert.run(provider.id, provider.name, provider.status);
    }
  });
  seed();
}

export function getAllProviders(): Provider[] {
  initializeProviders();
  return getDatabase()
    .prepare("SELECT * FROM providers WHERE id IN ('omniroute', 'freellmapi') ORDER BY CASE id WHEN 'omniroute' THEN 0 ELSE 1 END")
    .all() as Provider[];
}

export function getProviderById(id: string): Provider | undefined {
  initializeProviders();
  if (!isSupportedProviderId(id)) return undefined;
  return getDatabase()
    .prepare('SELECT * FROM providers WHERE id = ?')
    .get(id) as Provider | undefined;
}

export function markProviderReadyAfterVerification(id: SupportedProviderId): Provider | undefined {
  const provider = getProviderById(id);
  if (!provider) return undefined;
  getDatabase().prepare(`
    UPDATE providers
    SET status = 'ready', updated_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `).run(id);
  return getProviderById(id);
}

export function setProviderStatus(
  id: string,
  status: Exclude<ProviderStatus, 'ready'>,
): Provider | undefined {
  const provider = getProviderById(id);
  if (!provider) return undefined;
  getDatabase().prepare(`
    UPDATE providers
    SET status = ?,
        enabled = CASE WHEN ? = 'disabled' THEN 0 ELSE enabled END,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `).run(status, status, id);
  return getProviderById(id);
}

export function setProviderEnabled(id: string, enabled: boolean): Provider | undefined {
  const provider = getProviderById(id);
  if (!provider) return undefined;
  if (enabled && provider.status !== 'ready') throw new Error('PROVIDER_NOT_READY');
  getDatabase().prepare(`
    UPDATE providers SET enabled = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?
  `).run(enabled ? 1 : 0, id);
  return getProviderById(id);
}

import { getDatabase } from '../database.js';

export type SettingValue = string | number | boolean | object | null;

export function getSetting<T = SettingValue>(
  key: string,
  fallback: T | null = null
): T | null {
  const row = getDatabase()
    .prepare('SELECT value FROM settings WHERE key = ?')
    .get(key) as { value: string } | undefined;

  if (!row) return fallback;

  try {
    return JSON.parse(row.value) as T;
  } catch {
    return fallback;
  }
}

export function getAllSettings(): Record<string, SettingValue> {
  const rows = getDatabase()
    .prepare('SELECT key, value FROM settings')
    .all() as { key: string; value: string }[];

  const settings: Record<string, SettingValue> = {};

  for (const row of rows) {
    try {
      settings[row.key] = JSON.parse(row.value) as SettingValue;
    } catch {
      settings[row.key] = row.value;
    }
  }

  return settings;
}

export function setSetting(
  key: string,
  value: SettingValue
): void {
  getDatabase().prepare(`
    INSERT INTO settings (key, value, updated_at)
    VALUES (?, ?, CURRENT_TIMESTAMP)
    ON CONFLICT(key) DO UPDATE SET
      value = excluded.value,
      updated_at = CURRENT_TIMESTAMP
  `).run(key, JSON.stringify(value));
}

export function setMultipleSettings(
  settings: Record<string, SettingValue>
): void {
  const db = getDatabase();

  const save = db.transaction(() => {
    for (const [key, value] of Object.entries(settings)) {
      db.prepare(`
        INSERT INTO settings (key, value, updated_at)
        VALUES (?, ?, CURRENT_TIMESTAMP)
        ON CONFLICT(key) DO UPDATE SET
          value = excluded.value,
          updated_at = CURRENT_TIMESTAMP
      `).run(key, JSON.stringify(value));
    }
  });

  save();
}

export function deleteSetting(key: string): boolean {
  const result = getDatabase()
    .prepare('DELETE FROM settings WHERE key = ?')
    .run(key);

  return result.changes > 0;
}

export const SCHEMA_SQL = `
CREATE TABLE IF NOT EXISTS threads (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL DEFAULT 'New Chat',
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS messages (
  id TEXT PRIMARY KEY,
  thread_id TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('user', 'assistant', 'system')),
  content TEXT NOT NULL,
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (thread_id) REFERENCES threads(id) ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS idx_messages_thread_created
ON messages(thread_id, created_at);

CREATE TABLE IF NOT EXISTS settings (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL,
  updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS providers (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'unconfigured'
    CHECK (status IN ('unconfigured', 'pending_verification', 'ready', 'disabled')),
  enabled INTEGER NOT NULL DEFAULT 0 CHECK (enabled IN (0, 1)),
  config TEXT NOT NULL DEFAULT '{}',
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS chat_requests (
  request_id TEXT PRIMARY KEY,
  thread_id TEXT NOT NULL,
  user_message_id TEXT NOT NULL,
  assistant_message_id TEXT,
  provider_id TEXT,
  model TEXT,
  status TEXT NOT NULL CHECK (status IN ('running', 'completed')),
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (thread_id) REFERENCES threads(id) ON DELETE CASCADE,
  FOREIGN KEY (user_message_id) REFERENCES messages(id) ON DELETE CASCADE,
  FOREIGN KEY (assistant_message_id) REFERENCES messages(id) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS chat_requests (
  request_id TEXT PRIMARY KEY,
  thread_id TEXT NOT NULL,
  user_message_id TEXT NOT NULL,
  assistant_message_id TEXT,
  provider_id TEXT,
  model TEXT,
  status TEXT NOT NULL CHECK (status IN ('running', 'completed')),
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (thread_id) REFERENCES threads(id) ON DELETE CASCADE,
  FOREIGN KEY (user_message_id) REFERENCES messages(id) ON DELETE CASCADE,
  FOREIGN KEY (assistant_message_id) REFERENCES messages(id) ON DELETE SET NULL
);

CREATE VIRTUAL TABLE IF NOT EXISTS messages_fts USING fts5(
  message_id UNINDEXED,
  thread_id UNINDEXED,
  content,
  tokenize = 'unicode61'
);

CREATE TRIGGER IF NOT EXISTS messages_fts_insert
AFTER INSERT ON messages
BEGIN
  INSERT INTO messages_fts(message_id, thread_id, content)
  VALUES (new.id, new.thread_id, new.content);
END;

CREATE TRIGGER IF NOT EXISTS messages_fts_delete
AFTER DELETE ON messages
BEGIN
  DELETE FROM messages_fts WHERE message_id = old.id;
END;

CREATE TRIGGER IF NOT EXISTS messages_fts_update
AFTER UPDATE OF content ON messages
BEGIN
  DELETE FROM messages_fts WHERE message_id = old.id;
  INSERT INTO messages_fts(message_id, thread_id, content)
  VALUES (new.id, new.thread_id, new.content);
END;
`;

  res.setHeader('X-Accel-Buffering', 'no');
  res.flushHeaders();

  const sendEvent = (event: string, data: unknown) => {
    if (res.writableEnded) return;
    res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`);
  };

  try {
    const result = await sendChatMessage({
      ...parsed.data,
      signal: controller.signal,
      onToken: (token) => sendEvent('token', { token }),
    });
    sendEvent('done', {
      userMessage: result.userMessage,
      assistantMessage: result.assistantMessage,
      providerId: result.providerId,
      model: result.model,
    });
  } catch (error) {
    if (error instanceof ChatServiceError) {
      sendEvent('error', { error: error.code });
    } else if (error instanceof ProviderRequestError) {
      sendEvent('error', { error: error.code, providerId: error.providerId });
    } else {
      sendEvent('error', { error: 'CHAT_REQUEST_FAILED' });
    }
  } finally {
    if (!res.writableEnded) res.end();
  }
});

import { getProviderById, type Provider } from '../db/repositories/providerRepository.js';
import { getProviderAdapter } from './providerRegistry.js';
import type { ProviderGenerationRequest, ProviderGenerationResponse } from './providerAdapter.js';
import { ProviderRequestError } from './providerAdapter.js';

export type BrainProviderId = 'omniroute' | 'freellmapi';

function otherProvider(id: BrainProviderId): BrainProviderId {
  return id === 'omniroute' ? 'freellmapi' : 'omniroute';
}

function getReadyProvider(id: BrainProviderId): { provider: Provider; adapter: NonNullable<ReturnType<typeof getProviderAdapter>> } {
  const provider = getProviderById(id);
  const adapter = getProviderAdapter(id);
  if (!provider || !adapter || provider.status !== 'ready' || provider.enabled !== 1) {
    throw new ProviderRequestError(id, 'PROVIDER_NOT_READY', undefined, true);
  }
  return { provider, adapter };
}

function canFailover(error: unknown): boolean {
  return error instanceof ProviderRequestError && error.recoverable;
}

function getProviderOrder(selected?: string): BrainProviderId[] {
  const primary = selected === 'freellmapi' ? 'freellmapi' : 'omniroute';
  return [primary, otherProvider(primary)];
}

export async function generateWithFailover(
  request: ProviderGenerationRequest,
  selectedProvider?: string,
): Promise<ProviderGenerationResponse & { providerId: BrainProviderId }> {
  let lastError: unknown;
  const order = getProviderOrder(selectedProvider);

  for (let index = 0; index < order.length; index += 1) {
    const providerId = order[index];
    try {
      const { adapter } = getReadyProvider(providerId);
      const result = await adapter.generate(request);
      return { ...result, providerId };
    } catch (error) {
      lastError = error;
      if (!canFailover(error) || index === order.length - 1) throw error;
    }
  }

  throw lastError ?? new ProviderRequestError('omniroute', 'NO_PROVIDER_AVAILABLE');
}

export async function streamWithFailover(
  request: ProviderGenerationRequest,
  selectedProvider: string | undefined,
  onToken: (token: string) => void,
): Promise<ProviderGenerationResponse & { providerId: BrainProviderId }> {
  let lastError: unknown;
  const order = getProviderOrder(selectedProvider);

  for (let index = 0; index < order.length; index += 1) {
    const providerId = order[index];
    let emittedToken = false;
    try {
      const { adapter } = getReadyProvider(providerId);
      const result = await adapter.stream(request, (token) => {
        emittedToken = true;
        onToken(token);
      });
      return { ...result, providerId };
    } catch (error) {
      lastError = error;
      // Never switch providers after sending partial text to the client.
      if (emittedToken || !canFailover(error) || index === order.length - 1) throw error;
    }
  }

  throw lastError ?? new ProviderRequestError('omniroute', 'NO_PROVIDER_AVAILABLE');
}

export async function verifyProvider(providerId: BrainProviderId): Promise<string[]> {
  const adapter = getProviderAdapter(providerId);
  if (!adapter) throw new ProviderRequestError(providerId, 'PROVIDER_ADAPTER_NOT_REGISTERED');
  return adapter.verify();
}

  model?: string;

}

export class ChatServiceError extends Error {
  constructor(public readonly code: string, message = code) {
    super(message);
    this.name = 'ChatServiceError';
  }
}

export async function sendChatMessage(
  params: SendChatMessageParams,
): Promise<SendChatMessageResult> {
  const threadId = params.threadId.trim();
  const content = params.content.trim();
  const clientMessageId = params.clientMessageId.trim();

  if (!threadId) {
    throw new ChatServiceError('THREAD_ID_REQUIRED');
  }

  if (!content) {
    throw new ChatServiceError('MESSAGE_CONTENT_REQUIRED');
  }

  if (!clientMessageId || clientMessageId.length > 80) {
    throw new ChatServiceError('CLIENT_MESSAGE_ID_REQUIRED');
  }

  if (!getThreadById(threadId)) {
    throw new ChatServiceError('THREAD_NOT_FOUND');
  }

  // Claim the request BEFORE creating the user message.
  const claim = claimChatRequest(
    clientMessageId,
    threadId,
    clientMessageId,
  );

  if (!claim.claimed) {
    if (claim.record.thread_id !== threadId) {
      throw new ChatServiceError('CLIENT_MESSAGE_ID_CONFLICT');
    }

    if (
      claim.record.status === 'completed' &&
      claim.record.assistant_message_id
    ) {
      const userMessage = getMessageById(
        claim.record.user_message_id,
      );

      const assistantMessage = getMessageById(
        claim.record.assistant_message_id,
      );

      if (
        userMessage &&
        assistantMessage &&
        claim.record.provider_id
      ) {
        return {
          userMessage,
          assistantMessage,
          providerId: claim.record.provider_id as BrainProviderId,
          model: claim.record.model ?? undefined,
          replayed: true,
        };
      }
    }

    throw new ChatServiceError(
      'CHAT_REQUEST_ALREADY_IN_PROGRESS',
    );
  }

  try {
    // Create the user message only after successfully claiming.
    const userMessage = createMessage(
      threadId,
      'user',
      content,
      clientMessageId,
    );

    const history = getMessagesByThread(threadId).slice(-21);

    const messages: ChatMessage[] = history.map((message) => ({
      role: message.role,
      content: message.content,
    }));

    const response = await streamWithFailover(
      {
        messages,
        signal: params.signal,
      },
      params.providerId,
      params.onToken,
    );

    const assistantMessage = createMessage(
      threadId,
      'assistant',
      response.content,
    );

    completeChatRequest(
      clientMessageId,
      assistantMessage.id,
      response.providerId,
      response.model,
    );

    return {
      userMessage,
      assistantMessage,
      providerId: response.providerId,
      model: response.model,
    };
  } catch (error) {
    releaseChatRequest(clientMessageId);
    throw error;
  }
}

import type {
  ChatMessage,
  ProviderAdapter,
  ProviderGenerationRequest,
  ProviderGenerationResponse,
  ProviderUsage,
} from './providerAdapter.js';
import { ProviderRequestError } from './providerAdapter.js';

interface AdapterOptions {
  providerId: 'omniroute' | 'freellmapi';
  baseUrl: string;
  apiKey: () => string | undefined;
  defaultModel: () => string | undefined;
  timeoutMs: number;
}

interface ChatCompletionPayload {
  model?: string;
  choices?: Array<{
    message?: { content?: string | Array<{ type?: string; text?: string }> };
    delta?: { content?: string | Array<{ type?: string; text?: string }> };
    finish_reason?: string | null;
  }>;
  usage?: {
    prompt_tokens?: number;
    completion_tokens?: number;
    total_tokens?: number;
  };
  error?: { message?: string; code?: string };
}

interface RequestSignals {
  signal: AbortSignal;
  timeoutSignal: AbortSignal;
  parentSignal?: AbortSignal;
}

function contentToText(value: unknown): string {
  if (typeof value === 'string') return value;

  if (!Array.isArray(value)) return '';

  return value
    .map((part) => {
      if (
        part &&
        typeof part === 'object' &&
        'text' in part &&
        typeof part.text === 'string'
      ) {
        return part.text;
      }

      return '';
    })
    .join('');
}

function mapUsage(
  usage?: ChatCompletionPayload['usage'],
): ProviderUsage | undefined {
  if (!usage) return undefined;

  return {
    promptTokens: usage.prompt_tokens,
    completionTokens: usage.completion_tokens,
    totalTokens: usage.total_tokens,
  };
}

export class OpenAICompatibleAdapter implements ProviderAdapter {
  readonly providerId: 'omniroute' | 'freellmapi';

  private readonly baseUrl: string;
  private readonly getApiKey: () => string | undefined;
  private readonly getDefaultModel: () => string | undefined;
  private readonly timeoutMs: number;

  constructor(options: AdapterOptions) {
    this.providerId = options.providerId;
    this.baseUrl = options.baseUrl.replace(/\/+$/, '');
    this.getApiKey = options.apiKey;
    this.getDefaultModel = options.defaultModel;
    this.timeoutMs = options.timeoutMs;
  }

  private credentials(): { key: string; model: string } {
    const key = this.getApiKey()?.trim();
    const model = this.getDefaultModel()?.trim();

    if (!key) {
      throw new ProviderRequestError(
        this.providerId,
        'API_KEY_NOT_CONFIGURED',
        undefined,
        true,
      );
    }

    return {
      key,
      model: model ?? '',
    };
  }

  private createRequestSignals(parentSignal?: AbortSignal): RequestSignals {
    const timeoutSignal = AbortSignal.timeout(this.timeoutMs);

    return {
      signal: parentSignal
        ? AbortSignal.any([parentSignal, timeoutSignal])
        : timeoutSignal,
      timeoutSignal,
      parentSignal,
    };
  }

  private normalizeError(
    error: unknown,
    signals: RequestSignals,
    fallbackCode: string,
  ): ProviderRequestError {
    if (error instanceof ProviderRequestError) {
      return error;
    }

    // User cancellation must never trigger automatic failover.
    if (signals.parentSignal?.aborted) {
      return new ProviderRequestError(
        this.providerId,
        'REQUEST_ABORTED',
      );
    }

    // Timeout is recoverable and may use the backup provider.
    if (signals.timeoutSignal.aborted) {
      return new ProviderRequestError(
        this.providerId,
        'PROVIDER_TIMEOUT',
        undefined,
        true,
      );
    }

    return new ProviderRequestError(
      this.providerId,
      fallbackCode,
      undefined,
      true,
    );
  }

  private async fetchWithTimeout(
    path: string,
    init: RequestInit,
    signals: RequestSignals,
    apiKey: string,
  ): Promise<Response> {
    try {
      return await fetch(`${this.baseUrl}${path}`, {
        ...init,
        headers: {
          Authorization: `Bearer ${apiKey}`,
          ...(init.headers ?? {}),
        },
        signal: signals.signal,
      });
    } catch (error) {
      throw this.normalizeError(
        error,
        signals,
        'PROVIDER_CONNECTION_FAILED',
      );
    }
  }

  private async assertResponse(response: Response): Promise<void> {
    if (response.ok) return;

    const recoverable =
      response.status === 408 ||
      response.status === 429 ||
      response.status >= 500;

    const code =
      response.status === 401 || response.status === 403
        ? 'PROVIDER_AUTHENTICATION_FAILED'
        : response.status === 429
          ? 'PROVIDER_RATE_LIMITED'
          : response.status >= 500
            ? 'PROVIDER_SERVER_ERROR'
            : `PROVIDER_HTTP_${response.status}`;

    throw new ProviderRequestError(
      this.providerId,
      code,
      response.status,
      recoverable,
    );
  }

  async verify(): Promise<string[]> {
    const { key } = this.credentials();
    const signals = this.createRequestSignals();

    try {
      const response = await this.fetchWithTimeout(
        '/models',
        { method: 'GET' },
        signals,
        key,
      );

      await this.assertResponse(response);

      const payload = (await response.json()) as {
        data?: Array<{ id?: string }>;
      };

      return (payload.data ?? [])
        .map((item) => item.id)
        .filter((id): id is string => Boolean(id));
    } catch (error) {
      throw this.normalizeError(
        error,
        signals,
        'PROVIDER_VERIFICATION_FAILED',
      );
    }
  }

  async generate(
    request: ProviderGenerationRequest,
  ): Promise<ProviderGenerationResponse> {
    const { key, model: defaultModel } = this.credentials();

    if (!request.model && !defaultModel) {
      throw new ProviderRequestError(
        this.providerId,
        'MODEL_NOT_CONFIGURED',
        undefined,
        true,
      );
    }

    const signals = this.createRequestSignals(request.signal);

    try {
      const response = await this.fetchWithTimeout(
        '/chat/completions',
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            model: request.model ?? defaultModel,
            messages: request.messages,
            temperature: request.temperature,
            max_tokens: request.maxTokens,
            stream: false,
          }),
        },
        signals,
        key,
      );

      await this.assertResponse(response);

      const payload = (await response.json()) as ChatCompletionPayload;
      const content = contentToText(
        payload.choices?.[0]?.message?.content,
      );

      if (!content) {
        throw new ProviderRequestError(
          this.providerId,
          'PROVIDER_EMPTY_RESPONSE',
          undefined,
          true,
        );
      }

      return {
        content,
        model: payload.model,
        usage: mapUsage(payload.usage),
      };
    } catch (error) {
      throw this.normalizeError(
        error,
        signals,
        'PROVIDER_REQUEST_FAILED',
      );
    }
  }

  async stream(
    request: ProviderGenerationRequest,
    onToken: (token: string) => void,
  ): Promise<ProviderGenerationResponse> {
    const { key, model: defaultModel } = this.credentials();

    if (!request.model && !defaultModel) {
      throw new ProviderRequestError(
        this.providerId,
        'MODEL_NOT_CONFIGURED',
        undefined,
        true,
      );
    }

    const signals = this.createRequestSignals(request.signal);

    let reader: ReadableStreamDefaultReader<Uint8Array> | undefined;

    try {
      const response = await this.fetchWithTimeout(
        '/chat/completions',
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            Accept: 'text/event-stream',
          },
          body: JSON.stringify({
            model: request.model ?? defaultModel,
            messages: request.messages,
            temperature: request.temperature,
            max_tokens: request.maxTokens,
            stream: true,
          }),
        },
        signals,
        key,
      );

      await this.assertResponse(response);

      if (!response.body) {
        throw new ProviderRequestError(
          this.providerId,
          'PROVIDER_STREAM_UNAVAILABLE',
          undefined,
          true,
        );
      }

      reader = response.body.getReader();

      const decoder = new TextDecoder();

      let buffer = '';
      let content = '';
      let responseModel: string | undefined;
      let usage: ProviderUsage | undefined;

      const consumeEvent = (event: string): void => {
        const data = event
          .split('\n')
          .filter((line) => line.startsWith('data:'))
          .map((line) => line.slice(5).trim())
          .join('\n');

        if (!data || data === '[DONE]') return;

        let payload: ChatCompletionPayload;

        try {
          payload = JSON.parse(data) as ChatCompletionPayload;
        } catch {
          return;
        }

        if (payload.error) {
          throw new ProviderRequestError(
            this.providerId,
            'PROVIDER_STREAM_ERROR',
            undefined,
            true,
          );
        }

        responseModel = payload.model ?? responseModel;
        usage = mapUsage(payload.usage) ?? usage;

        const token = contentToText(
          payload.choices?.[0]?.delta?.content,
        );

        if (token) {
          content += token;
          onToken(token);
        }
      };

      while (true) {
        const { done, value } = await reader.read();

        if (done) break;

        buffer += decoder
          .decode(value, { stream: true })
          .replace(/\r\n/g, '\n');

        let boundary = buffer.indexOf('\n\n');

        while (boundary >= 0) {
          const event = buffer.slice(0, boundary);
          buffer = buffer.slice(boundary + 2);

          consumeEvent(event);

          boundary = buffer.indexOf('\n\n');
        }
      }

      buffer += decoder.decode();

      if (buffer.trim()) {
        consumeEvent(buffer);
      }

      if (!content) {
        throw new ProviderRequestError(
          this.providerId,
          'PROVIDER_EMPTY_RESPONSE',
          undefined,
          true,
        );
      }

      return {
        content,
        model: responseModel,
        usage,
      };
    } catch (error) {
      await reader?.cancel().catch(() => undefined);

      throw this.normalizeError(
        error,
        signals,
        'PROVIDER_STREAM_INTERRUPTED',
      );
    } finally {
      reader?.releaseLock();
    }
  }
}

import type { ProviderAdapter } from './providerAdapter.js';

const adapters = new Map<string, ProviderAdapter>();

export function registerProviderAdapter(
  adapter: ProviderAdapter
): void {
  const providerId = adapter.providerId.trim();

  if (!providerId) {
    throw new Error('PROVIDER_ID_REQUIRED');
  }

  if (adapters.has(providerId)) {
    throw new Error('PROVIDER_ADAPTER_ALREADY_REGISTERED');
  }

  adapters.set(providerId, adapter);
}

export function getProviderAdapter(
  providerId: string
): ProviderAdapter | undefined {
  return adapters.get(providerId.trim());
}

export function hasProviderAdapter(providerId: string): boolean {
  return adapters.has(providerId.trim());
}

export function unregisterProviderAdapter(providerId: string): boolean {
  return adapters.delete(providerId.trim());
}

export function getRegisteredProviderIds(): string[] {
  return Array.from(adapters.keys());
}

const PROVIDER_SECRET_ENV: Record<SecretProviderId, string> = {
  omniroute: 'OMNIROUTE_API_KEY',
  freellmapi: 'FREELLMAPI_API_KEY',
};

/**
 * Reads a provider API key from the process environment.
 * Secrets are never persisted to SQLite or returned by API routes.
 */
export function getProviderApiKey(
  providerId: SecretProviderId
): string | undefined {
  const envName = PROVIDER_SECRET_ENV[providerId];
  const value = process.env[envName];

  if (typeof value !== 'string' || value.length === 0) {
    return undefined;
  }

  return value;
}

/**
 * Checks whether a provider API key is configured.
 * Does not reveal or return the secret itself.
 */
export function hasProviderApiKey(
  providerId: SecretProviderId
): boolean {
  return getProviderApiKey(providerId) !== undefined;
}

async function api<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await fetch(`${API}${path}`, {
    ...init,
    headers: { 'Content-Type': 'application/json', ...(init?.headers ?? {}) },
  });
  const body = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(body.error || `HTTP_${response.status}`);
  return body as T;
}

    const content = input.trim();
    if (!content || busy || !threadId) return;
    const clientMessageId = crypto.randomUUID();
    const userMessage: Message = { id: clientMessageId, role: 'user', content };
    const assistantId = `pending-${clientMessageId}`;
    setMessages((current) => [...current, userMessage, { id: assistantId, role: 'assistant', content: '' }]);
    setInput('');
    setBusy(true);
    setTab('chat');

    try {
      const response = await fetch(`${API}/chat/stream`, {
        method: 'POST',

      });
      if (!response.ok || !response.body) {
        const body = await response.json().catch(() => ({}));
        throw new Error(body.error || `HTTP_${response.status}`);
      }
      const reader = response.body.getReader();
      const decoder = new TextDecoder();
      let buffer = '';
      let finished = false;
      while (!finished) {
        const { done, value } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');
        let boundary = buffer.indexOf('\n\n');
        while (boundary >= 0) {
          const eventText = buffer.slice(0, boundary);
          buffer = buffer.slice(boundary + 2);
          const eventName = eventText.split('\n').find((line) => line.startsWith('event:'))?.slice(6).trim();
          const dataText = eventText.split('\n').filter((line) => line.startsWith('data:')).map((line) => line.slice(5).trim()).join('\n');
          if (dataText) {

            if (eventName === 'token' && data.token) {

            if (eventName === 'done') {
              if (data.assistantMessage) {
                setMessages((current) => current.map((message) => message.id === assistantId ? data.assistantMessage! : message));
              }
              finished = true;
            }
          }
          boundary = buffer.indexOf('\n\n');
        }

import { useRef, useState } from 'react';
import { motion } from 'framer-motion';
import { Send, Mic, Paperclip } from 'lucide-react';
import type { CSSProperties, KeyboardEvent } from 'react';
import type { Message, ZaraStatus } from '../types';
import MessageBubble from './MessageBubble';

interface Props {
  messages: Message[];
  setMessages: React.Dispatch<React.SetStateAction<Message[]>>;
  status: ZaraStatus;
  setStatus: (status: ZaraStatus) => void;
}

export default function ChatInterface({
  messages,
  setMessages,
  status,
  setStatus,
}: Props) {
  const [input, setInput] = useState('');
  const bottomRef = useRef<HTMLDivElement>(null);

  const sendMessage = () => {
    const text = input.trim();
    if (!text || status === 'thinking') return;

    const userMessage: Message = {
      id: crypto.randomUUID(),
      role: 'user',
      content: text,
      timestamp: new Date(),
    };

    setMessages((previous) => [...previous, userMessage]);
    setInput('');

    // AI backend এখনো সংযুক্ত নয়।
    // এটি শুধু UI পরীক্ষা করার জন্য একটি demo reply।
    setStatus('thinking');

    window.setTimeout(() => {
      const demoMessage: Message = {
        id: crypto.randomUUID(),
        role: 'zara',
        content: 'আমি ZARA। আমার AI brain এখনো সংযুক্ত করা হয়নি। Backend যুক্ত হলে আমি তোমার প্রশ্নের উত্তর দিতে পারব।',
        timestamp: new Date(),
      };

      setMessages((previous) => [...previous, demoMessage]);
      setStatus('idle');
      bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
    }, 1200);
  };

  const handleKeyDown = (event: KeyboardEvent<HTMLTextAreaElement>) => {
    if (event.key === 'Enter' && !event.shiftKey) {
      event.preventDefault();
      sendMessage();
    }
  };

  return (
    <div style={styles.container}>
      <div style={styles.messages}>
        {messages.length === 0 && (
          <div style={styles.welcome}>
            <h2>How can I help you?</h2>
            <p>I'm ZARA. Your personal AI assistant.</p>
          </div>
        )}

        {messages.map((message) => (
          <MessageBubble key={message.id} message={message} />
        ))}

        {status === 'thinking' && (
          <div style={styles.thinking}>ZARA is thinking...</div>
        )}

        <div ref={bottomRef} />
      </div>

      <div style={styles.inputArea}>
        <div style={styles.inputBox}>
          <button type="button" style={styles.iconButton} aria-label="Attachment (not available yet)">
            <Paperclip size={19} />
          </button>

          <textarea
            value={input}
            onChange={(event) => setInput(event.target.value)}
            onKeyDown={handleKeyDown}
            placeholder="Message ZARA..."
            rows={1}
            style={styles.textarea}
          />

          <button
            type="button"
            style={styles.iconButton}
            aria-label="Voice input (not available yet)"
            disabled
          >
            <Mic size={19} />
          </button>

          <motion.button
            whileTap={{ scale: 0.92 }}
            type="button"
            onClick={sendMessage}
            disabled={!input.trim() || status === 'thinking'}
            style={{
              ...styles.sendButton,
              opacity: !input.trim() || status === 'thinking' ? 0.45 : 1,
            }}
            aria-label="Send message"
          >
            <Send size={18} />
          </motion.button>
        </div>

        <p style={styles.note}>
          AI connection is not configured yet. Voice and attachments are coming later.
        </p>
      </div>
    </div>
  );
}
